/** Run once: prints the real RAM of every daemon next to your free RAM, so you can see what fits. @param {NS} ns */
export async function main(ns) {
    const names = ["/hack/batcher.js", "purchase-server-8gb.js", "buy-programs.js", "upgrade-home-ram.js", "gang.js", "hacknet.js",
        "auto-backdoor.js", "backdoor-watch.js", "contracts.js", "IPvGO-lite.js", "darkweb.js", "sing-info.js", "orchestrator.js"];
    ns.tprint(`Home RAM ${ns.format.ram(ns.getServerMaxRam("home"))}, singularity info: ${ns.read("sing-info.txt") || "not written yet"}`);
    for (const n of names) {
        const r = ns.getScriptRam(n, "home");
        ns.tprint(`  ${n.padEnd(26)} ${r > 0 ? ns.format.ram(r) : "not on home"}`);
    }
}
