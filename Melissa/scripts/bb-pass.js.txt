// Bladeburner, ONE short pass (the orchestrator runs it every ~30 s once bb-state.txt says "run"): picks the best action for
// rank per second and starts it. The game keeps repeating the action between passes, so a pass only changes it when needed.
// Goal: rank 400000 and every Black Op, which lets auto-destroy.js end the BitNode (no hacking needed).
// Logic (no config): stamina first (rest below 55 %, until 90 %), then the next Black Op when its chance is safe, otherwise the
// Contract/Operation with the best chance x rank gain / seconds; Field Analysis when the estimates are too fuzzy,
// Diplomacy only when chaos is above 10, otherwise Training when nothing is safe yet. Writes bb-status.txt (read it: cat bb-status.txt).
const CONTRACTS = ["Tracking", "Bounty Hunter", "Retirement"];
const OPERATIONS = ["Investigation", "Undercover Operation", "Sting Operation", "Raid", "Stealth Retirement Operation", "Assassination"];
const MIN_CHANCE = 0.6;        // Contracts/Operations at or above this (lower estimate) are safe to run
const FALLBACK_CHANCE = 0.4;   // when nothing is that safe, the best one with at least this (middle estimate) beats idling
const BLACKOP_CHANCE = 0.8;    // a failed Black Op costs rank and health: wait for a safe chance
const REST_BELOW = 0.55, REST_UNTIL = 0.9;
const GENERAL = "General";

/** @param {NS} ns */
export async function main(ns) {
    ns.disableLog("ALL");
    const bb = ns.bladeburner;
    if (!bb.inBladeburner()) return;
    const [stam, maxStam] = bb.getStamina();
    const ratio = maxStam > 0 ? stam / maxStam : 1;
    const current = bb.getCurrentAction();   // { type, name } or null
    const rank = bb.getRank();
    const notes = [];

    const start = (type, name, why) => {
        if (current && current.type === type && current.name === name) { notes.push(`keeping ${name}: ${why}`); return true; }
        if (bb.startAction(type, name)) { notes.push(`started ${name}: ${why}`); return true; }
        return false;
    };

    let done = false;
    // 1. Stamina: low stamina makes every action fail more, so rest first
    const resting = current && current.name === "Hyperbolic Regeneration Chamber";
    if (ratio < REST_BELOW || (resting && ratio < REST_UNTIL)) done = start(GENERAL, "Hyperbolic Regeneration Chamber", `stamina ${Math.round(ratio * 100)}%`);

    // 2. The next Black Op, when rank allows and it is safe
    let next = null;
    if (!done) {
        next = bb.getNextBlackOp();
        if (next && rank >= next.rank) {
            const [lo, hi] = bb.getActionEstimatedSuccessChance("Black Operations", next.name);
            if (hi - lo > 0.1) done = start(GENERAL, "Field Analysis", `Black Op estimate too fuzzy (${Math.round(lo * 100)}-${Math.round(hi * 100)}%)`);
            else if (lo >= BLACKOP_CHANCE) done = start("Black Operations", next.name, `${Math.round(lo * 100)}% chance, rank ${Math.floor(rank)} >= ${next.rank}`);
            else notes.push(`${next.name} waits: only ${Math.round(lo * 100)}% chance (need ${BLACKOP_CHANCE * 100}%)`);
        }
    }

    // 3. Contracts and Operations by expected rank per second
    if (!done) {
        const rows = [];
        for (const [type, names] of [["Contracts", CONTRACTS], ["Operations", OPERATIONS]]) {
            for (const name of names) {
                try {
                    const [lo, hi] = bb.getActionEstimatedSuccessChance(type, name);
                    const gain = bb.getActionRankGain(type, name), secs = bb.getActionTime(type, name) / 1000;
                    const mid = (lo + hi) / 2;
                    let level = 1, maxLevel = 1, count = 0;
                    try { level = bb.getActionCurrentLevel(type, name); maxLevel = bb.getActionMaxLevel(type, name); count = bb.getActionCountRemaining(type, name); } catch { }
                    rows.push({ type, name, lo, hi, mid, gain, secs, level, maxLevel, count, score: (mid * gain) / Math.max(1, secs) });
                } catch (e) { notes.push(`${name}: not available (${String(e).slice(0, 60)})`); }
            }
        }
        rows.sort((a, b) => b.score - a.score);
        // Live readings for node-eta.js's Bladeburner model (route-model.js scales these to other skill levels with the game's formulas)
        try {
            const p = ns.getPlayer(), cities = [];
            for (const c of ["Sector-12", "Aevum", "Chongqing", "New Tokyo", "Ishima", "Volhaven"]) { try { cities.push({ name: c, pop: bb.getCityEstimatedPopulation(c), chaos: bb.getCityChaos(c) }); } catch { } }
            ns.write("bb-actions.json", JSON.stringify({ t: Date.now(), lv: { ...p.skills }, city: bb.getCity(), stamina: [stam, maxStam], cities,
                rows: rows.map(r => ({ type: r.type, name: r.name, chance: r.mid, lo: r.lo, hi: r.hi, gain: r.gain, secs: r.secs, level: r.level, maxLevel: r.maxLevel, count: r.count })) }), "w");
        } catch { }
        for (const r of rows.slice(0, 4)) notes.push(`  ${r.name}: ${Math.round(r.lo * 100)}-${Math.round(r.hi * 100)}% chance, ${r.gain.toFixed(2)} rank in ${r.secs.toFixed(0)} s => ${r.score.toFixed(3)} rank/s`);
        const good = rows.filter(r => r.lo >= MIN_CHANCE);                 // safe by the pessimistic estimate
        const okay = rows.filter(r => r.mid >= FALLBACK_CHANCE);            // acceptable by the middle estimate
        const pool = good.length ? good : okay;
        const spread = Math.max(0, ...rows.map(r => r.hi - r.lo));
        let chaos = 0;
        try { chaos = bb.getCityChaos(bb.getCity()); } catch { }

        if (pool.length && spread > 0.15 && good.length === 0) {
            done = start(GENERAL, "Field Analysis", `estimates too fuzzy to trust (up to ${Math.round(spread * 100)} points wide) and nothing is clearly safe yet`);
        } else if (pool.length) {
            const best = pool[0];
            const cur = current && pool.find(r => r.type === current.type && r.name === current.name);
            if (cur && cur.score >= best.score * 0.75) { notes.push(`keeping ${cur.name}: still within 25% of the best (${best.name})`); done = true; }
            else for (const r of pool) { if (start(r.type, r.name, `${Math.round(r.lo * 100)}-${Math.round(r.hi * 100)}% chance, ${r.score.toFixed(3)} rank/s`)) { done = true; break; } else notes.push(`${r.name}: startAction refused (none left?)`); }
        }
        if (!done && chaos > 10) done = start(GENERAL, "Diplomacy", `no action is safe and city chaos is ${chaos.toFixed(1)}`);
        if (!done && spread > 0.15) done = start(GENERAL, "Field Analysis", `no action is safe and estimates are fuzzy (${Math.round(spread * 100)} points)`);
        if (!done) done = start(GENERAL, "Training", `no Contract or Operation is safe yet (best ${rows[0] ? Math.round(rows[0].mid * 100) + "%" : "n/a"}, chaos ${chaos.toFixed(1)}): training raises the stats and so the chances`);
    }

    // Rank per hour, measured over this run, for node-eta.js
    let samples = [];
    try { samples = JSON.parse(ns.read("bb-rank.json")); } catch { }
    const now = Date.now();
    samples = samples.filter(s => now - s.t < 60 * 60 * 1000 && s.rank <= rank); samples.push({ t: now, rank });
    ns.write("bb-rank.json", JSON.stringify(samples), "w");
    const first = samples[0];
    const perHour = now - first.t > 120000 ? (rank - first.rank) / ((now - first.t) / 3600000) : NaN;
    ns.write("bb-status.txt", `${new Date().toLocaleTimeString()} rank ${Math.floor(rank)}` +
        `${Number.isFinite(perHour) ? ` (${Math.round(perHour)}/h)` : ""}; stamina ${Math.round(ratio * 100)}%; next Black Op: ${next ? `${next.name} at rank ${next.rank}` : "all done"}\n${notes.join("\n")}\n`, "w");
    if (ns.getHostname() !== "home") ns.scp(["bb-status.txt", "bb-rank.json", "bb-actions.json"].filter(f => ns.fileExists(f)), "home");
}
