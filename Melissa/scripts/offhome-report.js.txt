/** Run once: shows where each daemon runs, free RAM on every rooted server, and the RAM reserved for placements. @param {NS} ns */
export async function main(ns) {
    const seen = new Set(["home"]), queue = ["home"], hosts = [];
    while (queue.length) for (const next of ns.scan(queue.shift())) {
        if (seen.has(next)) continue;
        seen.add(next); queue.push(next);
        if (ns.hasRootAccess(next) && ns.getServerMaxRam(next) > 0) hosts.push(next);
    }
    const daemons = ["hacknet.js", "IPvGO-lite.js", "upgrade-home-ram.js", "buy-programs.js", "auto-backdoor.js", "purchase-server-8gb.js", "/hack/batcher.js", "gang.js", "contracts.js"];
    for (const d of daemons) {
        const where = ["home", ...hosts].filter(h => ns.isRunning(d, h));
        ns.tprint(`${d.padEnd(24)} ${where.length ? where.join(", ") : "not running"}`);
    }
    const raw = ns.peek(40);
    ns.tprint(`Reserved on other servers (port 40): ${raw === "NULL PORT DATA" ? "none" : raw}`);
    const rows = hosts.map(h => ({ h, max: ns.getServerMaxRam(h), free: ns.getServerMaxRam(h) - ns.getServerUsedRam(h) }))
        .sort((a, b) => b.max - a.max);
    ns.tprint(`${hosts.length} rooted servers with RAM (max / free GB):`);
    for (const r of rows.slice(0, 15)) ns.tprint(`  ${r.h.padEnd(22)} ${r.max} / ${r.free.toFixed(1)}`);
}
