
import { loadConfig } from "/utils/config.js";
import { wantedBoosts } from "/utils/aug-filter.js";
// Port player-work.js reads the goals from (20 darknet, 21 stock influence, 22 batcher income).
export const GOALS_PORT = 23;
const NEUROFLUX = "NeuroFlux Governor";

// "Worth it" is the shared rule in utils/aug-filter.js (same as buy-augs.js and
// singularity/workers/city-plan.js), driven by `augments` in the config (config.js, data/config.json).

/**
 * Works out the real rep goal of each faction passed as an argument (in priority order):
 * the highest rep requirement among the augments still worth buying there, i.e. not owned
 * and not already bought (queued). getAugmentationRepReq includes the BitNode's rep-cost
 * multiplier. An augment sold by several of the factions counts only for the first joined
 * one in the list (or the first one, if none is joined), as before.
 *
 * Started by singularity/player-work.js; writes
 *   { time, goals: { faction: { rep, aug, count } } }
 * to port 23 and exits. About 22GB, but only for a moment.
 */
/** @param {NS} ns */
export async function main(ns) {
	// Config order first (the preference), then EVERY other joined faction: a faction you are in is never ignored just because it is not listed
	const cfgOrder = loadConfig(ns).factionWork.order;
	const factions = ns.args.length ? ns.args.map(String) : [...new Set([...cfgOrder, ...ns.getPlayer().factions])];
	const augConfig = loadConfig(ns).augments;
	const joined = new Set(ns.getPlayer().factions);
	const owned = new Set(ns.singularity.getOwnedAugmentations(true)); // true: bought-but-not-installed count too
	const order = [...factions.filter((faction) => joined.has(faction)), ...factions.filter((faction) => !joined.has(faction))];

	const goals = Object.fromEntries(factions.map((faction) => [faction, { rep: 0, aug: "", count: 0 }]));
	const claimed = new Set();
	const candidates = [];
	for (const faction of order) {
		for (const aug of ns.singularity.getAugmentationsFromFaction(faction)) {
			if (aug === NEUROFLUX || owned.has(aug) || claimed.has(aug) || !wantedBoosts(aug, ns.singularity.getAugmentationStats(aug), augConfig)) continue;
			claimed.add(aug);
			candidates.push({ faction, aug, rep: ns.singularity.getAugmentationRepReq(aug), price: ns.singularity.getAugmentationPrice(aug) });
		}
	}

	// Changed from the repo: cost is considered. Rep is only ground for augments you can expect to afford: the cash you have plus
	// what the batcher earns in `rep-horizon.txt` (hours, set with horizon.js), else `repGrindHorizonHours` (config), default 2, counting that every purchase raises the price of the
	// rest by 1.9x (cheapest first are kept while the whole set, bought most expensive first, still fits). Without a fresh
	// income reading (port 24, posted by the batcher) there is no cost filter.
	const posted = ns.peek(24);
	const income = posted && typeof posted === "object" && Number.isFinite(posted.perSecond) && Date.now() - posted.time <= 5 * 60_000 ? posted.perSecond : null;
	const budget = ns.getServerMoneyAvailable("home") + (income ?? 0) * (Number(ns.read("rep-horizon.txt")) || loadConfig(ns).repGrindHorizonHours || 2) * 3600;
	let kept = candidates;
	if (income !== null) {
		const setCost = (list) => [...list].sort((a, b) => b.price - a.price).reduce((sum, c, i) => sum + c.price * Math.pow(1.9, i), 0);
		kept = [];
		for (const c of [...candidates].sort((a, b) => a.price - b.price)) {
			if (c.price === 0 || setCost([...kept, c]) <= budget) kept.push(c);
		}
	}
	for (const c of kept) {
		const goal = goals[c.faction];
		goal.count++;
		if (c.rep > goal.rep) Object.assign(goal, { rep: c.rep, aug: c.aug });
	}

	ns.clearPort(GOALS_PORT);
	ns.writePort(GOALS_PORT, JSON.stringify({ time: Date.now(), goals, budget, income, skippedForCost: candidates.length - kept.length }));
}
