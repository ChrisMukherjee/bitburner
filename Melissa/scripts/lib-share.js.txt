// Shared settings for RAM sharing, used by orchestrator.js, share.js and hack/batcher.js.
// Sharing only boosts reputation while you work for a faction, so it is off until you run: run share.js on

export const DEFAULT_SHARE_PERCENT = 25;     // Share of home RAM used by "run share.js on" without a number
export const SHARE_FLAG = "share-on.txt";    // Exists on home while sharing is on; holds the chosen percent
export const SHARE_WORKER = "share-worker.js";
export const KEEP_FREE_CAP_GB = 64;           // Most GB kept free on home
export const KEEP_FREE_SHARE = 0.25;          // Share of home RAM kept free (capped), so small homes still run their daemons

/** GB the batcher and share worker leave free on home for the other daemons: 25% of home RAM, at most 64 GB. */
export function homeReserveGb(ns) {
    return Math.min(KEEP_FREE_CAP_GB, Math.floor(ns.getServerMaxRam("home") * KEEP_FREE_SHARE));
}

/** Percent of home RAM sharing may use, or 0 when sharing is off. */
export function sharePercent(ns) {
    if (!ns.fileExists(SHARE_FLAG, "home")) return 0;
    const pct = Number(ns.read(SHARE_FLAG));
    return pct > 0 && pct <= 100 ? pct : DEFAULT_SHARE_PERCENT;
}

/** Threads the share worker should run on home (0 when sharing is off). */
export function shareThreadsWanted(ns) {
    const ram = ns.getScriptRam(SHARE_WORKER, "home");
    if (ram <= 0) return 0;
    return Math.floor(ns.getServerMaxRam("home") * sharePercent(ns) / 100 / ram);
}

/** GB on home the batcher should leave free so the share worker can reach its full size. */
export function shareRamToReserve(ns) {
    const ram = ns.getScriptRam(SHARE_WORKER, "home");
    const running = ns.getRunningScript(SHARE_WORKER, "home");
    return Math.max(0, (shareThreadsWanted(ns) - (running ? running.threads : 0)) * ram);
}
