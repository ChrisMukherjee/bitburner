/**
 * One weaken. args: target, delay, land, tol
 * `land` = the exact time (ms since epoch) the weaken must land. It reads its own duration at the moment it starts,
 * so it lands on time even if started a little early or late and even while security is raised.
 * A weaken that is late is still run.
 * Without `land`, `delay` = extra ms to wait (old style).
 * Kept as three tiny scripts (hack, grow, weaken) because a script's RAM counts every ns function it mentions.
 * @param {NS} ns */
export async function main(ns) {
    const target = ns.args[0];
    let extra = Math.max(0, Math.floor(Number(ns.args[1]) || 0));
    const land = Number(ns.args[2]) || 0;
    const tol = Number(ns.args[3]) || 0;
    if (land > 0) {
        extra = Math.floor(land - Date.now() - ns.getWeakenTime(target));
        extra = Math.max(0, extra); // a weaken is never skipped, being late only leaves security up a little longer
    }
    await ns.weaken(target, { additionalMsec: extra });
}
