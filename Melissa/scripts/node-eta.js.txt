const competence = (a, lv) => STATS.reduce((t, s, i) => t + a.w[i] * Math.pow(Math.max(1, lv[s] ?? 1), a.dc[i]), 0);
const timeFactor = (lv) => 0.5 * (Math.pow(lv.agility, 0.04) + Math.pow(lv.dexterity, 0.035) + lv.agility / 1e4 + lv.dexterity / 1e4);

/**
 * Bladeburner rank model. `rows` are the live readings written by bb-pass.js (bb-actions.json): per action the chance, level, count,
 * seconds and rank gain, plus `lv` = the skill levels at that moment. The chance of an action at other skill levels / action levels is
 * scaled from the live reading with the game's own formula (competence = sum of weight x level^decay; difficulty = base x fac^(level-1)),
 * so everything not modelled (skill points, city population, stamina) stays as measured.
 */
function makeBladeburner(rows) {
    const acts = [];
    for (const r of rows) {
        const a = BY_NAME.get(r.name); if (!a) continue;
        const lvl = Math.max(1, r.level || 1), maxLevel = Math.max(lvl, r.maxLevel || lvl);
        const need = (m) => Math.ceil(0.5 * m * (2 * (a.contract ? 3 : 2.5) + m - 1));
        acts.push({ a, ref: r, level: maxLevel, maxLevel, succ: need(maxLevel - 1), count: Math.max(0, r.count ?? 1), cap: a.contract ? 150 : 100,
            compRef: competence(a, r.lv), tfRef: timeFactor(r.lv), need });
    }
    return acts;
}
const bladeburnerClone = (acts) => acts.map(x => ({ ...x }));

function actionNow(x, lv) {
    const { a, ref } = x;
    const dRatio = Math.pow(a.dfac, x.level - Math.max(1, ref.level || 1));
    const chance = Math.min(1, ref.chance * (competence(a, lv) / x.compRef) / dRatio);
    const secs = Math.max(1, ref.secs * dRatio * timeFactor(ref.lv) / timeFactor(lv));
    const gain = ref.gain * Math.pow(a.rfac, x.level - Math.max(1, ref.level || 1));
    return { chance, secs, gain };
}

/** Advance the model `hours` at skill levels `lv`; policy as bb-pass.js: best chance x gain / seconds among actions with a safe-ish chance. */
function bladeburnerAdvance(acts, state, lv, hours, duty, minChance = 0.4, dt = 0.25) {
    for (let t = 0; t < hours - 1e-9; t += dt) {
        const h = Math.min(dt, hours - t);
        let best = null;
        for (const x of acts) {
            if (x.count < 1) continue;
            const n = actionNow(x, lv);
            if (n.chance < minChance) continue;
            const score = n.chance * n.gain / n.secs;
            if (!best || score > best.score) best = { x, n, score };
        }
        for (const x of acts) x.count = Math.min(x.cap, x.count + x.a.growth * h * 3600 / 480);
        if (!best) continue;
        const { x, n } = best;
        const attempts = Math.min(x.count, h * 3600 * duty / n.secs);
        x.count -= attempts;
        const wins = attempts * n.chance;
        x.succ += wins; state.rank += wins * n.gain;
        while (x.succ >= x.need(x.maxLevel)) { x.maxLevel++; x.level = x.maxLevel; }
    }
}

/** Best action's chance now (the live "run is worth it" figure), from the live rows rescaled to the current skill levels. */
function bestChanceNow(acts, lv) {
    let best = 0, name = null;
    for (const x of acts) { const n = actionNow(x, lv); if (n.chance > best) { best = n.chance; name = x.a.name; } }
    return { chance: best, name };
}

/** Rank per hour the model predicts at skill levels `lv` for `hours` of play (duty 1), used to calibrate against the measured rate. */
function predictRankPerHour(acts, lv, hours = 1, minChance = 0.4) {
    const c = bladeburnerClone(acts), st = { rank: 0 };
    bladeburnerAdvance(c, st, lv, hours, 1, minChance);
    return st.rank / hours;
}

/**
 * Multi-run projection. goal "combat": buy combat augs each run, retrain the combat stats in the gym, then play Bladeburner until rank 400000.
 * goal "hacking": buy hacking augs each run until Daedalus/The Red Pill and then 9000 hacking within one run are reachable.
 * Scenarios differ only in how income and exp rate grow with the hacking multiplier (low: not at all; high: income x mult, exp x sqrt(mult)).
 *
 * Faction rep (v3.0.1 formulas): hacking work gives 5 cycles/s x (hacking level / 975) x faction_rep mult x BitNode rep mult per second, so it
 * grows with the level reached in the run. Rep persists across installs per faction. An augmentation needs rep with a faction that sells it. Once a
 * faction's favour is 150 (cumulative rep 25000 x (1.02^150 - 1), about 450k) rep can also be bought: $1e6 per rep / (faction_rep x BitNode mult).
 *
 * cfg: { augs: [{ name, price, rep, f: [factions], h, e, cs[4] }], nf, mult: {hacking, hackExp, combat[4], combatExp}, income, expRate, installed,
 *        runHours, overheadHours, rep: { facMult, bn, share }, scenario: {income, exp}, maxRuns, bb: { acts, rank, lv, duty, gymHours } | null,
 *        hack: { required, daedalusAugs, daedalusMoney, daedalusHack, rpRep, rpPrice, rpHeld, startStage } }
 */
const DONATE_REP = 25000 * (Math.pow(1.02, 150) - 1);
function projectRoute(goal, cfg) {
    const maxRuns = cfg.maxRuns ?? 60, T = cfg.runHours, over = cfg.overheadHours ?? 0.25;
    const m = { hacking: cfg.mult.hacking, hackExp: cfg.mult.hackExp, combat: [...cfg.mult.combat] };
    const mult0 = { ...m };
    const rm = (cfg.rep?.facMult ?? 1) * (cfg.rep?.bn ?? 1), share = cfg.rep?.share ?? 0.7;
    const pool = cfg.augs.map(a => ({ ...a, f: a.f && a.f.length ? a.f : ["?"] }));
    const Rf = {};                                   // faction -> cumulative rep (persists across installs)
    const R = (f) => Rf[f] ?? 0, maxR = (a) => Math.max(0, ...a.f.map(R)), allMax = () => Math.max(0, ...Object.values(Rf));
    let hours = 0, installed = cfg.installed ?? 0, state = { rank: cfg.bb?.rank ?? 0 }, acts = cfg.bb ? bladeburnerClone(cfg.bb.acts) : null;
    let nfLevel = 0;
    let stage = cfg.hack?.startStage ?? 0;           // hacking: 0 = before Daedalus, 1 = Red Pill rep/price, 2 = Red Pill installed
    const log = [];
    const value = (a) => goal === "combat" ? a.cs.reduce((t, v) => t + Math.log(Math.max(1, v)), 0) / 4 : Math.log(Math.max(1, a.h)) + 0.1 * Math.log(Math.max(1, a.e));
    for (let k = 1; k <= maxRuns; k++) {
        const income = cfg.income * Math.pow(Math.max(1, m.hacking / mult0.hacking), cfg.scenario.income);
        const budget = income * T * 3600 * 0.5;       // income ramps up during a run: about half of T x income is spendable
        const rate = cfg.expRate * (m.hackExp / mult0.hackExp) * Math.pow(Math.max(1, m.hacking / mult0.hacking), cfg.scenario.exp);   // hacking exp/s this run
        const levelAt = (hrs) => skillAt(rate * hrs * 3600, m.hacking);
        // Rep earned this run: hacking work at the level reached a quarter of the way into the run
        const repRun = 5 * (levelAt(T * 0.25) / 975) * rm * T * 3600 * share;
        let repLeft = repRun;
        if (goal === "hacking" && stage === 1) { Rf.Daedalus = R("Daedalus") + repRun; repLeft = 0; }
        for (let guard = 0; guard < 80 && repLeft > 1; guard++) {
            let best = null;
            for (const a of pool) {
                if (a.taken || value(a) <= 0.002 || maxR(a) >= a.rep) continue;
                const f = a.f.reduce((x, y) => R(y) > R(x) ? y : x, a.f[0]), deficit = a.rep - R(f);
                const sc = value(a) / deficit;
                if (!best || sc > best.sc) best = { a, f, deficit, sc };
            }
            if (!best) break;
            const give = Math.min(repLeft, best.deficit);
            Rf[best.f] = R(best.f) + give; repLeft -= give;
            if (give < best.deficit) break;
        }
        if (repLeft > 1) { const top = Object.keys(Rf).sort((x, y) => R(y) - R(x))[0] ?? "?"; Rf[top] = R(top) + repLeft; }
        let spent = 0, n = 0; const bought = [];
        const cands = pool.filter(a => !a.taken && value(a) > 0.002).map(a => {
            const f = a.f.reduce((x, y) => R(y) > R(x) ? y : x, a.f[0]), deficit = Math.max(0, a.rep - R(f));
            const donate = deficit > 0 && R(f) >= DONATE_REP;            // favour 150: rep can be bought
            return { a, f, deficit, donate, ok: deficit === 0 || donate, extra: donate ? deficit * 1e6 / rm : 0 };
        }).filter(c => c.ok).sort((x, y) => value(y.a) / (y.a.price + y.extra) - value(x.a) / (x.a.price + x.extra));
        for (const c of cands) {
            const p = c.a.price * Math.pow(AUG_STEP, n) + c.extra;
            if (spent + p > budget) continue;
            spent += p; n++; c.a.taken = true; bought.push(c.a.name);
            if (c.deficit > 0) Rf[c.f] = c.a.rep;
            m.hacking *= Math.max(1, c.a.h); m.hackExp *= Math.max(1, c.a.e);
            for (let i = 0; i < 4; i++) m.combat[i] *= Math.max(1, c.a.cs[i]);
        }
        // NeuroFlux Governor: every level multiplies all skills a little (compounding), costs x1.14 more per level (price and rep), and counts in the x1.9 step
        if (cfg.nf) {
            for (let guard = 0; guard < 400; guard++) {
                const p = cfg.nf.price * Math.pow(1.14, nfLevel) * Math.pow(AUG_STEP, n), r = cfg.nf.rep * Math.pow(1.14, nfLevel);
                if (spent + p > budget || r > allMax()) break;
                spent += p; n++; nfLevel++; bought.push("NeuroFlux Governor");
                m.hacking *= Math.max(1, cfg.nf.h); m.hackExp *= Math.max(1, cfg.nf.e);
                for (let i = 0; i < 4; i++) m.combat[i] *= Math.max(1, cfg.nf.cs[i]);
            }
        }
        installed += bought.length;
        let runHours = T;
        const entry = { run: k, names: bought, bought: bought.length, hackMult: m.hacking, combat: [...m.combat], income, repRun, spent, topRep: allMax() };
        if (goal === "hacking") {
            const H = cfg.hack;
            const rate2 = cfg.expRate * (m.hackExp / mult0.hackExp) * Math.pow(Math.max(1, m.hacking / mult0.hacking), cfg.scenario.exp);
            const level2 = (hrs) => skillAt(rate2 * hrs * 3600, m.hacking);
            if (stage === 0) {
                // Daedalus needs N augmentations installed, hacking level, and the money in hand at once: a long run if only the money is missing
                if (installed >= H.daedalusAugs && level2(T) >= H.daedalusHack) {
                    const needH = H.daedalusMoney / (income * 0.8) / 3600;
                    if (needH <= 36) { runHours = Math.max(T, needH); stage = 1; entry.note = "Daedalus invite"; }
                }
            } else if (stage === 1) {
                const d = R("Daedalus"), repOk = d >= H.rpRep || H.rpHeld, buyRep = !repOk && d >= DONATE_REP ? (H.rpRep - d) * 1e6 / rm : (repOk ? 0 : Infinity);
                if (Number.isFinite(buyRep) && budget >= H.rpPrice + buyRep) { stage = 2; entry.note = "Red Pill bought"; }
            } else if (stage === 2) {
                // The Red Pill is installed at the start of this run: can this run reach the required level?
                const need = expFor(H.required, m.hacking) / rate2 / 3600;
                if (need <= 36) { runHours = Math.max(need, 0.5); hours += runHours; entry.note = "final run"; entry.hours = hours; log.push(entry); return { reached: true, hours, runs: k, log }; }
            }
            if (level2(T) >= H.required && stage < 2) entry.levelOnlyNote = "level reachable but The Red Pill is not installed yet";
        } else {
            // Gym retrain: the install reset all skill exp; the four combat stats share gymHours
            const gym = cfg.bb.gymHours, E = GYM_EXP_PER_SECOND * 3600 * (gym / 4) * (cfg.mult.combatExp ?? 1);
            const lv = { ...cfg.bb.lv, strength: skillAt(E, m.combat[0]), defense: skillAt(E, m.combat[1]), dexterity: skillAt(E, m.combat[2]), agility: skillAt(E, m.combat[3]) };
            entry.lv = lv;
            const rank0 = state.rank;
            bladeburnerAdvance(acts, state, lv, Math.max(0, T - gym), cfg.bb.duty);
            entry.rank = state.rank;
            if (state.rank >= 400000) {
                const frac = Math.min(1, Math.max(0, (400000 - rank0) / Math.max(1e-9, state.rank - rank0)));
                hours += gym + Math.max(0, T - gym) * frac; entry.hours = hours; log.push(entry); return { reached: true, hours, runs: k, log };
            }
        }
        hours += runHours + over; entry.hours = hours; log.push(entry);
    }
    return { reached: false, hours, runs: maxRuns, log };
}

/** The game's chance before population, stamina, skill multipliers and the like: min(1, competence / difficulty) at an action level. */
function rawChance(name, lv, level = 1) {
    const a = BY_NAME.get(name); if (!a) return NaN;
    return Math.min(1, competence(a, lv) / (a.base * Math.pow(a.dfac, Math.max(1, level) - 1)));
}
/** Smallest income multiplier (1, 2, 4, ...) at which projectRoute(goal, cfg) reaches its goal within maxHours with flat income growth; Infinity if none up to 4096. */
function incomeNeeded(goal, cfg, maxHours = 100) {
    for (let k = 1; k <= 4096; k *= 2) {
        const r = projectRoute(goal, { ...cfg, income: cfg.income * k, scenario: { income: 0, exp: 0 } });
        if (r.reached && r.hours <= maxHours) return k;
    }
    return Infinity;
}

/**
 * Evaluates a faction invite requirement against live state: { cost: seconds of the player's time (Infinity = cannot be done by
 * this script's methods), todo: what has to happen }. Works for every PlayerRequirement type of the game, so no faction is hard-coded.
 */
function evalReq(r, c) {
    const done = { cost: 0, todo: [] };
    const miss = (cost, text) => ({ cost, todo: [text] });
    switch (r.type) {
        case "money": return c.money >= r.money ? done : miss(c.income > 0 ? (r.money - c.money) / c.income : Infinity, `have ${usd(r.money)}`);
        case "skills": {
            let cost = 0; const todo = [];
            for (const [k, v] of Object.entries(r.skills)) {
                if ((c.skills[k] ?? 0) >= v) continue;
                const per = k === "hacking" ? c.hackRate : GYM_EXP_PER_SECOND * (c.mults[k + "_exp"] || 1);
                const need = Math.max(0, expFor(v, c.mults[k] || 1) - (c.exp[k] ?? 0));
                cost += per > 0 ? need / per : Infinity; todo.push(`${k} ${c.skills[k]}->${v}`);
            }
            return todo.length ? miss(cost, `train ${todo.join(", ")}`) : done;
        }
        case "karma": {
            const ok = r.karma < 0 ? c.karma <= r.karma : c.karma >= r.karma;
            if (ok) return done;
            return miss(c.karmaRate > 0 ? Math.abs(c.karma - r.karma) / c.karmaRate : Infinity, `crime until karma ${r.karma} (now ${c.karma.toFixed(0)})`);
        }
        case "numPeopleKilled": return c.killed >= r.numPeopleKilled ? done : miss(c.killRate > 0 ? (r.numPeopleKilled - c.killed) / c.killRate : Infinity, `kill ${r.numPeopleKilled - c.killed} more (crime)`);
        case "file": return c.fileExists(r.file) ? done : miss(Infinity, `have file ${r.file}`);
        case "numAugmentations": return c.augCount >= r.numAugmentations ? done : miss(Infinity, `${r.numAugmentations} augmentations installed`);
        case "city": return c.city === r.city ? done : miss(0, `travel to ${r.city}`);
        case "location": return c.location === r.location ? done : miss(0, `be at ${r.location}`);
        case "backdoorInstalled": return c.backdoor(r.server) ? done : miss(Infinity, `backdoor ${r.server}`);
        case "bitNodeN": return c.node === r.bitNodeN ? done : miss(Infinity, `be in BitNode ${r.bitNodeN}`);
        case "sourceFile": return c.sf.has(r.sourceFile) ? done : miss(Infinity, `own Source-File ${r.sourceFile}`);
        case "bladeburnerRank": return c.rank >= r.bladeburnerRank ? done : miss(Infinity, `Bladeburner rank ${r.bladeburnerRank}`);
        case "not": return evalReq(r.condition, c).cost === 0 ? miss(Infinity, `NOT: ${r.condition.type} (a condition that blocks the invite)`) : done;
        case "someCondition": { let best = null; for (const x of r.conditions) { const e = evalReq(x, c); if (!best || e.cost < best.cost) best = e; } return best ?? done; }
        case "everyCondition": return r.conditions.map(x => evalReq(x, c)).reduce((a, e) => ({ cost: a.cost + e.cost, todo: a.todo.concat(e.todo) }), done);
        default: return miss(Infinity, `${r.type} (company/hacknet/infiltration work: not estimated)`);   // employedBy, jobTitle, companyReputation, hacknet*, numInfiltrations
    }
}

    let chanceNow = null, duty = 0.5, predicted = NaN, measured = NaN, calibrated = false, rankNow = 0, rph = NaN;
    if (bladeAccess && !bladeburnerDone) {
        try { const r = JSON.parse(ns.read("bb-rank.json")); const r0 = r[0], r1 = r[r.length - 1]; rankNow = r1.rank; if (r1.t - r0.t > 30 * 60000) { rph = (r1.rank - r0.rank) / ((r1.t - r0.t) / 3600000); calibrated = true; } else if (r1.t - r0.t > 120000) rph = (r1.rank - r0.rank) / ((r1.t - r0.t) / 3600000); } catch { }
        if (!rankNow) rankNow = rankSeen;
        if (actData && Array.isArray(actData.rows) && actData.rows.length) {
            acts = RM.makeBladeburner(actData.rows.map(r => ({ ...r, lv: actData.lv })));
            chanceNow = RM.bestChanceNow(acts, lvNow);
            const best0 = RM.bestChanceNow(acts, actData.lv).chance;
            predicted = RM.predictRankPerHour(acts, actData.lv, 1, Math.min(0.4, best0 * 0.9));
            measured = rph;
            if (predicted > 0 && Number.isFinite(measured)) duty = Math.min(1, Math.max(0.05, measured / predicted));
        }
        if (acts && income > 0) {
            const bbCfg = (gym) => ({ acts, rank: rankNow, lv: lvNow, duty, gymHours: gym });
            bbSim = scenarios.map(sc => {
                let best = null;
                for (const g of GYMS) {
                    const r = RM.projectRoute("combat", { augs: augList, nf, mult: { hacking: mult, hackExp: 1, combat: cNowM, combatExp: player.mults.strength_exp || 1 }, income, expRate: rate || 1,
                        installed: installedNow.size, runHours: RUN_HOURS, rep: repCfg, scenario: sc, maxRuns: 60, bb: bbCfg(g) });
                    r.gym = g;
                    if (!best || (r.reached && !best.reached) || (r.reached === best.reached && r.hours < best.hours)) best = r;
                }
                return { sc, r: best };
            });
        } else bbNote = !acts ? "no live Bladeburner readings yet (bb-actions.json appears after bb-pass.js has run once)" : "needs measured income";
    }

    // Income needed: with growth unknown, the robust question is "how much income would make this route a 100 h route" (flat-income projection)
    let needHack = NaN, needBB = NaN;
    if (hackSim) needHack = RM.incomeNeeded("hacking", { augs: augList, nf, mult: { hacking: mult, hackExp: player.mults.hacking_exp || 1, combat: cNowM, combatExp: 1 }, income, expRate: rate, installed: installedNow.size, runHours: RUN_HOURS, rep: repCfg, maxRuns: 60, hack: { required, daedalusAugs: 30, daedalusMoney: 1e11, daedalusHack: 2500, rpRep, rpPrice, rpHeld: haveRep >= rpRep, startStage: installed ? 2 : joined ? 1 : 0 } });
    if (bbSim) needBB = Math.min(...GYMS.map(g => RM.incomeNeeded("combat", { augs: augList, nf, mult: { hacking: mult, hackExp: 1, combat: cNowM, combatExp: player.mults.strength_exp || 1 }, income, expRate: rate || 1, installed: installedNow.size, runHours: RUN_HOURS, rep: repCfg, maxRuns: 60, bb: { acts, rank: rankNow, lv: lvNow, duty, gymHours: g } })));
    lines.push(`--- Assumptions: aug multipliers multiply and stay installed through every reset; skill level = mult x (32 ln(exp+534.5) - 200) with exp restarting at each install; ${RUN_HOURS} h per run + 0.25 h reset overhead; about half of income x run is spendable (it ramps up); each aug bought in a run costs x${RM.AUG_STEP} more; faction rep from the game's formula (hacking work, x${repCfg.facMult.toFixed(2)} faction_rep, x${repCfg.bn} BitNode, 70% of the run) with donations once a faction's favour reaches 150; pool of ${augList.length} hacking/combat augs`);
    lines.push(`Confidence: ${confidence}`);
    lines.push(`--- Route simulation (reset by reset)`);
    if (hackSim) {
        for (const { sc, r } of hackSim) {
            const last = r.log[r.log.length - 1];
            lines.push(`Hacking, ${sc.name}: ${fmtH(r)}${r.reached ? "" : `; after ${r.runs} runs hacking x${last.hackMult.toFixed(1)}`}`);
        }
        const first = hackSim[1]?.r.log[0];
        if (first && first.names.length) lines.push(`Hacking, what run 1 buys: ${first.names.slice(0, 8).join(", ")}${first.names.length > 8 ? ", ..." : ""} (hacking x${first.hackMult.toFixed(1)} after)`);
    } else if (!installed || level < required) lines.push(`Hacking: ${hackNote || "not simulated"}`);
    if (bbSim) {
        for (const { sc, r } of bbSim) {
            const last = r.log[r.log.length - 1];
            lines.push(`Bladeburner, ${sc.name}: ${fmtH(r)}${r.reached ? "" : `; after ${r.runs} runs combat x${last.combat.map(v => v.toFixed(0)).join("/")}, rank ${Math.round(last.rank || 0)}`} (gym ${r.gym} h per run)`);
        }
        const first = bbSim[1]?.r.log[0];
        if (first && first.names.length) lines.push(`Bladeburner, what run 1 buys: ${first.names.slice(0, 8).join(", ")}${first.names.length > 8 ? ", ..." : ""}`);
        lines.push(`Bladeburner calibration: the model predicts ${Number.isFinite(predicted) ? predicted.toFixed(1) : "?"} rank/h at full duty from the live readings; measured ${Number.isFinite(measured) ? measured.toFixed(1) : "?"} rank/h${calibrated ? "" : " (window under 30 min: not yet reliable)"} => duty ${Math.round(duty * 100)}% applied to the projections (stamina rest, retraining and idle time)`);
        lines.push(`Best action chance now ${Math.round((chanceNow?.chance ?? 0) * 100)}% (${chanceNow?.name ?? "n/a"}) at your current skill levels; rank ${Math.floor(rankNow)}/400000. Not modelled: skill points (help), Black Op chances and rank gates (hurt), city population (a better city raises every chance)`);
    } else if (bladeAccess && !bladeburnerDone) lines.push(`Bladeburner: ${bbNote || "not simulated"}`);

    if (hackSim || bbSim) lines.push(`Income that would make a route take about 100 h (flat-income model, today's $${ns.format.number(income)}/s): hacking ${Number.isFinite(needHack) ? "$" + ns.format.number(income * needHack) + "/s (x" + needHack + ")" : "more than x4096"}; Bladeburner ${Number.isFinite(needBB) ? "$" + ns.format.number(income * needBB) + "/s (x" + needBB + ")" : "more than x4096"}`);
    // Environment factor: live chance / the game's raw chance (population, stamina, skill multipliers); and what a better city would do
    if (acts && actData) {
        const trk = actData.rows.find(r => r.name === "Tracking") ?? actData.rows[0];
        const raw = RM.rawChance(trk.name, actData.lv, trk.level);
        const cities = Array.isArray(actData.cities) ? actData.cities.filter(c => c.pop > 0) : [];
        const popFac = (c) => Math.pow(c.pop / 1e9, 0.7) / (c.chaos > 50 ? Math.sqrt(1 + c.chaos - 50) : 1);
        const cur = cities.find(c => c.name === actData.city), top = cities.slice().sort((a, b) => popFac(b) - popFac(a))[0];
        lines.push(`Environment: ${trk.name} chance ${Math.round(trk.chance * 100)}% vs ${Math.round(raw * 100)}% from skills alone => other factors x${(trk.chance / Math.max(1e-9, raw)).toFixed(2)} (city population, stamina, skills)${cur && top ? `; city ${cur.name} factor ${popFac(cur).toFixed(2)}, best city ${top.name} ${popFac(top).toFixed(2)}` : ""}`);
    }
    // ---- Verdict: only calibrated numbers are used (over 400 h counts as "no estimate")
    const mid = (sim) => sim && sim[1] ? sim[1].r : null;
    const okH = (r) => r && r.reached && r.hours <= 400 ? r.hours : Infinity;
    const progress = (sim, key) => { const r = mid(sim); if (!r) return 0; const l = r.log[r.log.length - 1]; return key === "hack" ? l.hackMult : (l.rank || 0) / 400000 + Math.min(...l.combat) / 1e3; };
    const hH = okH(mid(hackSim)), bH = okH(mid(bbSim));
    let route = "";
    if (bladeburnerDone) route = "";
    else if (!bladeAccess) route = "hacking";
    else {
        // One route at a time. The choice is kept between runs of this script (bb-route.txt) and only changes when the other route is
        // clearly better by its own measure: hours when a scenario finishes, else the income it would need. Flipping wastes the gym time,
        // rank and faction rep already put into the committed route.
        const prev = String(ns.read("bb-route.txt")).trim();
        const anyH = Number.isFinite(hH) || Number.isFinite(bH);
        const sh = anyH ? hH : (Number.isFinite(needHack) ? needHack : Infinity), sb = anyH ? bH : (Number.isFinite(needBB) ? needBB : Infinity);
        const best = sh === Infinity && sb === Infinity ? (hackSim ? "hacking" : "combat") : sh <= sb ? "hacking" : "combat";
        route = best;
        if ((prev === "hacking" || prev === "combat") && prev !== best) {
            const keep = prev === "hacking" ? sh : sb, other = prev === "hacking" ? sb : sh;
            if (Number.isFinite(keep) && !(other < 0.75 * keep)) route = prev;   // switch only for a 25 % better route, or when the committed one can no longer finish
            if (!Number.isFinite(keep) && !Number.isFinite(other)) route = prev;
        }
    }
    const next = (() => {
        if (route === "hacking") { const f = hackSim?.[1]?.r.log[0]; return f && f.names.length ? `buy ${f.names.slice(0, 5).join(", ")} (the cheapest strong hacking augs), install, and let the batcher earn; repeat each run` : "keep the batcher earning; buy hacking augs as income allows"; }
        if (plan && plan.steps.length) { const st = plan.steps[0]; return st.inv.joined ? `work for ${st.f} until rep ${Math.round(Math.max(...st.pick.map(r => r.rep)))}, then buy ${st.pick.map(r => r.a).join(", ")}` : `${st.inv.todo.join("; ") || "accept the " + st.f + " invite"} to join ${st.f}, then buy ${st.pick.map(r => r.a).join(", ")}`; }
        return "keep earning; re-check when income rises";
    })();
    const anyNumber = Number.isFinite(hH) || Number.isFinite(bH);
    if (!bladeburnerDone) {
        lines.push(`RECOMMENDED ROUTE: ${route === "hacking" ? "hacking over several resets" : "Bladeburner (combat augs first)"}${anyNumber ? `, about ${Math.min(hH, bH).toFixed(0)} h in the middle scenario` : ": ETA unknown (no scenario finishes within 400 h at today's income); this is the route that needs the smaller income increase to become a 100 h route"}`);
        lines.push(`NEXT STEP: ${next}`);
    }
    // Is Bladeburner worth the player's time right now? Decided from the live chance rescaled to the current skills.
    const wasHold = String(ns.read("bb-hold.txt")).trim() === "1";
    const holdNow = !!(acts && chanceNow && chanceNow.chance < (wasHold ? 0.4 : 0.3));   // hysteresis: hold under 30 %, resume only above 40 %
    lines.push(`Bladeburner "run" now: ${!acts ? "no data yet: let it run once to read the real chances" : holdNow ? `HOLD: best chance ${Math.round(chanceNow.chance * 100)}% is too low to earn rank; the player's time goes to faction rep/earning instead (bb-manage.js pauses it and resumes by itself when the chance passes 30%)` : `run: best chance ${Math.round(chanceNow.chance * 100)}%`}`);
    if (bladeAccess) {
        ns.write("bb-hold.txt", holdNow ? "1" : "0", "w");
        // Commit a route only once both sides have been measured: with no Bladeburner readings yet (bb-actions.json), "hacking" would make
        // bb-manage.js hold Bladeburner, so bb-pass.js would never read the chances and the combat route could never be evaluated.
        // "" = no verdict: bb-manage.js trains and joins as before, bb-pass.js measures, and the next pass decides.
        ns.write("bb-route.txt", acts ? route : "", "w");
        if (!acts && route) lines.push(`Route not committed yet (bb-route.txt empty): Bladeburner has no live readings, so it trains, joins and measures first`);
    }
    const ready = (installed && level >= required) || bladeburnerDone;
    ns.write("node-ready.txt", ready ? "1" : "0", "w");
    ns.write("node-eta.txt", lines.join("\n") + "\n", "w");
    if (ns.getHostname() !== "home") ns.scp(["node-eta.txt", "node-ready.txt", "node-eta-state.json", "bb-train-target.txt", "bb-hold.txt", "bb-route.txt", "node-eta-history.json"].filter(f => ns.fileExists(f)), "home");
}

/** @param {NS} ns */
export async function main(ns) {
    ns.disableLog("ALL");
    const started = Date.now();
    const failWith = (what, e) => {
        const text = `${new Date().toLocaleTimeString()}  node-eta FAILED (${what}): ${String(e && e.stack ? e.stack : e).slice(0, 600)}\n` +
            `Host ${ns.getHostname()}. If this says the module cannot be found, if the game says it cannot resolve a module, restart the orchestrator (kill orchestrator.js, run orchestrator.js).\n`;
        ns.write("node-eta.txt", text, "w");
        if (ns.getHostname() !== "home") ns.scp("node-eta.txt", "home");
        ns.tprint("[node-eta] " + text.split("\n")[0]);
    };
    // Pull the data files from home itself (the module and income.js come with the orchestrator's files list: they are static imports)
    if (ns.getHostname() !== "home") {
        for (const f of ["bb-actions.json", "node-eta-history.json", "data/aug-plan.json", "bb-state.txt", "bb-status.txt", "bb-rank.json", "bb-route.txt"]) {
            try { if (ns.fileExists(f, "home")) ns.scp(f, ns.getHostname(), "home"); } catch { }
        }
    }
    try { await run(ns); } catch (e) { failWith("error while computing", e); await readyOnly(ns); return; }
    // Timing line at the end of the report, so a slow run is visible
    try { ns.write("node-eta.txt", String(ns.read("node-eta.txt")) + `(computed in ${Date.now() - started} ms)\n`, "w"); if (ns.getHostname() !== "home") ns.scp("node-eta.txt", "home"); } catch { }
}

/** Fallback when the report fails: still keep node-ready.txt right, because auto-destroy.js depends on it. */
async function readyOnly(ns) {
    try {
        if (!String(ns.read("sing-info.txt")).startsWith("1")) return;
        const installed = ns.singularity.getOwnedAugmentations(false).includes(RED_PILL);
        const node = ns.getResetInfo().currentNode;
        let required = 3000 * (WORLD_DAEMON_DIFFICULTY[node] ?? 1);
        try { required = ns.getServer(WORLD).requiredHackingSkill; } catch { }
        let done = false;
        done = /next Black Op: all done/.test(String(ns.read("bb-status.txt")).split("\n")[0]);
        ns.write("node-ready.txt", (installed && ns.getPlayer().skills.hacking >= required) || done ? "1" : "0", "w");
        if (ns.getHostname() !== "home") ns.scp("node-ready.txt", "home");
    } catch { }
}
