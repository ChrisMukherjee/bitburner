// One-shot: deletes the 28 retired scripts from the GAME's home server (the file sync cannot delete). Copies of all of them are
// kept in Bitburner-archive\retired on your PC. Touches only the files named below; missing ones are skipped.
//   run remove-retired.js            delete them
//   run remove-retired.js --dry      only list what would be deleted
const RETIRED = [
    "controller.js", "startup.js", "IPvGO.js", "IPvGOV2.js", "purge.js", "purge-intel.js", "contractSolver.js",
    "refresh-worker.js", "backdoor.js", "hacking_setup.js", "early-hack-template.js", "away.js", "clean.js",
    "cloud-status.js", "darkfind.js", "details.js", "dnetfuncs.js", "find.js", "find-backdoors.js", "phish.js",
    "pin.js", "stasis.js", "shareRam.js", "worker.js", "kill-workers.js", "contracts-selftest.js", "auth-report.js",
    "hack/controller-jit.js",
];

/** @param {NS} ns */
export async function main(ns) {
    const dry = ns.args.includes("--dry");
    const removed = [], missing = [], failed = [];
    for (const file of RETIRED) {
        if (!ns.fileExists(file, "home")) { missing.push(file); continue; }
        if (dry) { removed.push(file); continue; }
        if (ns.rm(file, "home")) removed.push(file); else failed.push(file);   // rm fails when the script is still running
    }
    ns.tprint(`${dry ? "Would remove" : "Removed"} ${removed.length}: ${removed.join(", ") || "none"}`);
    if (missing.length) ns.tprint(`Not in the game (skipped): ${missing.length}`);
    if (failed.length) ns.tprint(`Could not remove (still running?): ${failed.join(", ")}`);
}
