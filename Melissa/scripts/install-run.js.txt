// One-shot, started by the orchestrator after reset-now.js and buy-augs.js have run: installs the bought augmentations
// (this resets the run) and starts orchestrator.js on home afterwards. Does nothing if no augmentation was bought.
/** @param {NS} ns */
export async function main(ns) {
    if (!String(ns.read("sing-info.txt")).startsWith("1")) return;
    ns.rm("do-reset.txt", "home");
    const queued = ns.singularity.getOwnedAugmentations(true).length - ns.singularity.getOwnedAugmentations(false).length;
    if (queued <= 0) {
        ns.toast("Reset cancelled: no augmentation was bought, so there is nothing to install.", "warning", null);
        return;
    }
    // Extra cash goes into the home computer first: RAM, then cores (both are kept through the install)
    let upgrades = 0;
    while (Number.isFinite(ns.singularity.getUpgradeHomeRamCost()) && ns.singularity.upgradeHomeRam()) upgrades++;
    while (Number.isFinite(ns.singularity.getUpgradeHomeCoresCost()) && ns.singularity.upgradeHomeCores()) upgrades++;
    if (upgrades > 0) ns.tprint(`[reset] spent leftover cash on ${upgrades} home upgrade(s) before installing`);
    ns.singularity.installAugmentations("orchestrator.js");
}
