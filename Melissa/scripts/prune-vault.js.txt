import { loadVault, VAULT, FOUND } from 'lib-intel.js';

const BACKUP = "passwords-backup.txt";
const KEEP_WORDS = ["", "password", "admin", "102"];
const MIN_LIVE = 5;
const LAST_SIZE = "prune-map-size.txt";
const MAX_SHRINK = 0.3;

/** Trims passwords.json down to the darknet servers that exist right now.
 *  Run from home: run prune-vault.js          (dry run, only prints what would change)
 *                 run prune-vault.js --apply  (backs up, then writes the trimmed vault)
 *  orchestrator.js runs it with --auto (same as --apply, but logs quietly instead of to the terminal)
 *  @param {NS} ns */
export async function main(ns) {
  if (ns.getHostname() !== "home") {
    ns.tprint("ERROR: Run prune-vault.js from home.");
    return;
  }
  const auto = ns.args.includes("--auto");
  const apply = auto || ns.args.includes("--apply");
  const say = msg => auto ? ns.print(msg) : ns.tprint(msg);

  // Same crawl as darkweb.js: follow every scout's scout_nodes.txt
  const live = new Set(["darkweb"]);
  const queue = ["darkweb"];
  const scanned = new Set();
  while (queue.length > 0) {
    const current = queue.shift();
    if (scanned.has(current)) continue;
    scanned.add(current);
    try {
      if (!ns.fileExists("scout_nodes.txt", current)) continue;
      ns.scp("scout_nodes.txt", "home", current);
      for (const n of JSON.parse(ns.read("scout_nodes.txt") || "[]")) {
        live.add(n);
        if (!scanned.has(n)) queue.push(n);
      }
    } catch (e) {}
  }

  // If the scouts aren't running the map is empty, and pruning would wipe every key
  if (live.size < MIN_LIVE) {
    say(`ERROR: Only ${live.size} darknet servers mapped. Start darkweb.js and let the scouts run first.`);
    return;
  }

  // A sudden drop usually means a scout died and its part of the map is missing for now.
  // Skip this time; if the map is still that small next run, the drop is real and we prune.
  const lastSize = Number(ns.read(LAST_SIZE)) || 0;
  if (apply) ns.write(LAST_SIZE, String(live.size), "w");
  if (apply && live.size < lastSize * (1 - MAX_SHRINK)) {
    say(`[Prune] Map shrank from ${lastSize} to ${live.size} servers since last run. Skipping in case a scout is down.`);
    return;
  }

  // No awaits from here on, so darkweb.js can't write the vault halfway through
  const vault = loadVault(ns, VAULT);

  // Pull in keys the nodes found but home hasn't merged yet, so clearing their vault_found.txt loses nothing
  const foundOn = [];
  for (const host of live) {
    if (host === "home" || !ns.fileExists(FOUND, host)) continue;
    if (!ns.scp(FOUND, "home", host)) continue;
    const found = loadVault(ns, FOUND);
    for (const [h, key] of Object.entries(found.servers)) vault.servers[h] = key;
    for (const word of found.dictionary) if (!vault.dictionary.includes(word)) vault.dictionary.push(word);
    foundOn.push(host);
  }

  const servers = {};
  for (const [host, key] of Object.entries(vault.servers)) {
    if (live.has(host)) servers[host] = key;
  }

  // Keep the defaults, the keys of live servers, and any word that isn't just digits (hints and names).
  // Numeric words were one server's random passcode, and that server is gone.
  const liveKeys = new Set(Object.values(servers));
  const dictionary = vault.dictionary.filter(w =>
    KEEP_WORDS.includes(w) || liveKeys.has(w) || !/^\d+$/.test(w));

  const before = JSON.stringify(vault, null, 2);
  const pruned = { ...vault, servers, dictionary };
  const after = JSON.stringify(pruned, null, 2);

  say(`[Prune] ${live.size} darknet servers mapped right now.`);
  say(`[Prune] Servers: ${Object.keys(vault.servers).length} -> ${Object.keys(servers).length}`);
  say(`[Prune] Dictionary: ${vault.dictionary.length} -> ${dictionary.length}`);
  say(`[Prune] Size: ${(before.length / 1024).toFixed(1)} KB -> ${(after.length / 1024).toFixed(1)} KB`);
  say(`[Prune] vault_found.txt to clear on ${foundOn.length} nodes.`);

  if (!apply) {
    say("[Prune] Dry run, nothing changed. Run again with --apply to write it.");
    return;
  }

  ns.write(BACKUP, before, "w");
  ns.write(VAULT, after, "w");
  // Their keys are now in home's vault; darkweb.js copies that vault back out on its next loop
  for (const host of foundOn) ns.rm(FOUND, host);
  ns.rm(FOUND, "home");

  say(`[Prune] Done. Old vault saved as ${BACKUP}.`);
}
