import { SHARE_FLAG, DEFAULT_SHARE_PERCENT, sharePercent } from "lib-share.js";

/**
 * Switches RAM sharing on or off. orchestrator.js picks the change up within a minute.
 * run share.js on       - before working for a faction (sharing boosts the reputation you earn)
 * run share.js on 75    - same, but use up to 75% of home RAM (e.g. when you're away and only want rep)
 * run share.js off      - when you stop
 * @param {NS} ns
 */
export async function main(ns) {
    const mode = String(ns.args[0] ?? "").toLowerCase();
    if (mode === "on") {
        const pct = ns.args[1] === undefined ? DEFAULT_SHARE_PERCENT : Number(ns.args[1]);
        if (!(pct > 0 && pct <= 100)) {
            ns.tprint("[Share] The percent must be between 1 and 100, e.g. run share.js on 75");
            return;
        }
        ns.write(SHARE_FLAG, String(pct), "w");
        ns.tprint(`[Share] On at ${pct}% of home RAM. The orchestrator applies it within a minute or two.`);
    } else if (mode === "off") {
        ns.rm(SHARE_FLAG, "home");
        ns.tprint("[Share] Off. The orchestrator will stop sharing within a minute.");
    } else {
        const pct = sharePercent(ns);
        ns.tprint(`[Share] Sharing is ${pct ? `on at ${pct}%` : "off"}. Use: run share.js on [percent] | run share.js off`);
    }
}
